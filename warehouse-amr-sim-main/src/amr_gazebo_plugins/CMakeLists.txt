cmake_minimum_required(VERSION 3.8)
project(amr_gazebo_plugins)
if(NOT CMAKE_CXX_STANDARD)
  set(CMAKE_CXX_STANDARD 17)
endif()
find_package(ament_cmake REQUIRED)
find_package(gazebo_dev REQUIRED)
find_package(gazebo_ros REQUIRED)
find_package(rclcpp REQUIRED)
find_package(amr_interfaces REQUIRED)

add_library(amr_pedestrian_pose_plugin SHARED src/pedestrian_pose_plugin.cpp)
ament_target_dependencies(amr_pedestrian_pose_plugin gazebo_dev gazebo_ros rclcpp amr_interfaces)
install(TARGETS amr_pedestrian_pose_plugin LIBRARY DESTINATION lib)
ament_package()
